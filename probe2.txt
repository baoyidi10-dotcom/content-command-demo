/* ===================== utils ===================== */
function pad(n){return String(n).padStart(2,"0")}
function esc(s){return String(s==null?"":s).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;")}
function hasCJK(s){return /[\u4e00-\u9fff\u3400-\u4dbf]/.test(String(s||""))}
function toISO(d){return d.getFullYear()+"-"+pad(d.getMonth()+1)+"-"+pad(d.getDate())}
function parseISO(s){var p=s.split("-").map(Number);return new Date(p[0],p[1]-1,p[2])}
function mondayOf(d){var x=new Date(d),g=x.getDay();x.setDate(x.getDate()+(g===0?-6:1-g));x.setHours(0,0,0,0);return x}
function addDays(d,n){var x=new Date(d);x.setDate(x.getDate()+n);return x}
function sameDay(a,b){return a.getFullYear()===b.getFullYear()&&a.getMonth()===b.getMonth()&&a.getDate()===b.getDate()}
function nowISO(){return new Date().toISOString()}
function weekNo(d){var t=new Date(d.getFullYear(),d.getMonth(),d.getDate());t.setDate(t.getDate()+3-((t.getDay()+6)%7));var w1=new Date(t.getFullYear(),0,4);return 1+Math.round(((t-w1)/86400000-3+((w1.getDay()+6)%7))/7)}
function fmtRange(s){var e=addDays(s,6);return s.toLocaleDateString("en-GB",{month:"short",day:"numeric"}).toUpperCase()+"  "+e.toLocaleDateString("en-GB",{month:"short",day:"numeric",year:"numeric"}).toUpperCase()}

var Particles=(function(){
  var cv,ctx,parts=[],running=true,raf;
  function resize(){cv.width=innerWidth;cv.height=innerHeight}
  function init(){
    cv=document.getElementById("particles");ctx=cv.getContext("2d");resize();
    addEventListener("resize",resize);
    for(var i=0;i<55;i++)parts.push({x:Math.random()*innerWidth,y:Math.random()*innerHeight,r:Math.random()*1.5+.5,vy:-(Math.random()*.25+.06),vx:(Math.random()-.5)*.12,a:Math.random()*.35+.08,tw:Math.random()*6.28});
    loop();
  }
  function loop(){
    if(!running){raf=null;return}
    raf=requestAnimationFrame(loop);ctx.clearRect(0,0,cv.width,cv.height);
    for(var i=0;i<parts.length;i++){var p=parts[i];p.y+=p.vy;p.x+=p.vx;p.tw+=.03;
      if(p.y<-5){p.y=cv.height+5;p.x=Math.random()*cv.width}
      var a=p.a*(.65+.35*Math.sin(p.tw));
      ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,6.2832);ctx.fillStyle="rgba(244,196,48,"+a.toFixed(3)+")";ctx.fill();}
  }
  return{init:init,setCalm:function(v){running=!v;if(running&&!raf)loop();if(!running&&raf){cancelAnimationFrame(raf);raf=null;ctx.clearRect(0,0,cv.width,cv.height)}}};
})();

var Beams=(function(){
  var cv,ctx,raf,running=true;
  var BEAMS=[{ax:-.12,ay:-.22,dir:38,half:7.5,T:23,ph:0},{ax:1.14,ay:-.20,dir:142,half:8.5,T:31,ph:1.7},{ax:-.10,ay:1.18,dir:-34,half:6,T:37,ph:3.1},{ax:1.10,ay:1.16,dir:-148,half:6.5,T:41,ph:4.4}];
  function resize(){cv.width=innerWidth;cv.height=innerHeight}
  function I(b,t){return .70+.30*(.62*Math.sin(2*Math.PI*t/b.T+b.ph)+.38*Math.sin(2*Math.PI*t/(b.T*1.618)+2*b.ph))}
  function TH(b,t){return (b.dir+1.4*Math.sin(2*Math.PI*t/(b.T*2.3)+b.ph))*Math.PI/180}
  function frame(ts){
    if(!running){raf=null;return}
    raf=requestAnimationFrame(frame);
    var t=ts/1000,W=cv.width,H=cv.height,DIAG=Math.sqrt(W*W+H*H);
    ctx.clearRect(0,0,W,H);
    for(var i=0;i<BEAMS.length;i++){
      var b=BEAMS[i],inten=I(b,t);
      var ox=b.ax*W,oy=b.ay*H,th=TH(b,t),L=1.7*DIAG,hw=b.half*Math.PI/180;
      var x1=ox+Math.cos(th-hw)*L,y1=oy+Math.sin(th-hw)*L;
      var x2=ox+Math.cos(th+hw)*L,y2=oy+Math.sin(th+hw)*L;
      var g=ctx.createLinearGradient(ox,oy,ox+Math.cos(th)*L,oy+Math.sin(th)*L);
      g.addColorStop(0,"rgba(244,196,48,"+(.07*inten).toFixed(4)+")");
      g.addColorStop(.45,"rgba(244,196,48,"+(.016*inten).toFixed(4)+")");
      g.addColorStop(1,"rgba(244,196,48,0)");
      ctx.fillStyle=g;ctx.beginPath();ctx.moveTo(ox,oy);ctx.lineTo(x1,y1);ctx.lineTo(x2,y2);ctx.closePath();ctx.fill();
    }
  }
  function init(){
    cv=document.getElementById("beams");ctx=cv.getContext("2d");resize();
    addEventListener("resize",resize);
    var reduce=matchMedia("(prefers-reduced-motion: reduce)").matches;
    if(reduce){running=false;return}
    raf=requestAnimationFrame(frame);
    document.addEventListener("visibilitychange",function(){
      if(document.hidden){if(raf){cancelAnimationFrame(raf);raf=null}}
      else if(!raf&&running){raf=requestAnimationFrame(frame)}
    });
  }
  return{init:init,setCalm:function(v){running=!v;if(running&&!raf)raf=requestAnimationFrame(frame);if(!running&&raf){cancelAnimationFrame(raf);raf=null;ctx.clearRect(0,0,cv.width,cv.height)}}};
})();

/* ===================== log ===================== */
function logit(kind,msg){
  var d=new Date(),ts=pad(d.getHours())+":"+pad(d.getMinutes())+":"+pad(d.getSeconds());
  logLines.push({ts:ts,k:kind,m:msg});
  if(logLines.length>3)logLines.shift();
  document.getElementById("log").innerHTML=logLines.map(function(l,i){
    return '<div class="'+(i===logLines.length-1?"fresh ":"")+(l.k==="REFUSED"?"ref":"")+'"><span class="k">'+esc(l.k)+"</span>"+esc(l.ts)+"  "+esc(l.m)+"</div>";
  }).join("");
}

function confetti(){
  for(var i=0;i<30;i++){
    var c=document.createElement("div");c.className="confetti";
    c.style.left=(45+Math.random()*10)+"%";c.style.top="40%";c.style.background=["#f4c430","#ffd700","#fff","#4FB3A8"][Math.floor(Math.random()*4)];
    document.body.appendChild(c);
    (function(el){
      var dx=(Math.random()-.5)*280,dy=-(Math.random()*220+60),rot=Math.random()*720;
      el.animate([{transform:"translate(0,0) rotate(0deg)",opacity:1},{transform:"translate("+dx+"px,"+dy+"px) rotate("+rot+"deg)",opacity:0}],{duration:900+Math.random()*500,easing:"cubic-bezier(.22,.61,.36,1)"}).onfinish=function(){el.remove()};
    })(c);
  }
}
